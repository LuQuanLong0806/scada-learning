一、核心数据类型(M0 Day 2 定义,几乎所有模块都用)

using System;

namespace DaqMonitor.Core.Models;

// 设备状态(值类型,适合状态机)
public enum DeviceState
{
    Offline,        // 离线
    Connecting,     // 连接中
    Online,         // 在线
    Error           // 异常
}

// 报警级别
public enum AlarmLevel
{
    Normal,         // 正常
    Warning,        // 预警
    Critical        // 严重
}

// 采集点位 - 上位机最基础的数据载体
// struct:值类型,小而频繁传递,GC 压力小(Day 2 学过)
public readonly struct SensorPoint
{
    public int Id { get; init; }            // 点位编号
    public double Value { get; init; }      // 实时数值
    public DateTime Time { get; init; }     // 采集时刻

    public SensorPoint(int id, double value, DateTime? time = null)
    {
        Id = id;
        Value = value;
        Time = time ?? DateTime.Now;
    }
}

// 历史记录(M4 持久化用,带主键 ID 便于 EF Core 落库)
public class SensorRecord
{
    public long Id { get; set; }            // 自增主键
    public int PointId { get; set; }        // 点位编号
    public double Value { get; set; }       // 数值
    public DateTime Time { get; set; }      // 时刻
}

// 报警事件
public class AlarmEvent
{
    public int PointId { get; init; }
    public double Value { get; init; }
    public AlarmLevel Level { get; init; }
    public DateTime Time { get; init; } = DateTime.Now;
    public string Message { get; init; } = "";
}

二、设备抽象(M0 Day 5 定义,M1/M2/M3/M11/M13 都继承)

using System.Threading;
using System.Threading.Tasks;

namespace DaqMonitor.Core.Devices;

// 设备统一接口 - "合同":任何设备都得能 Connect/Disconnect/Read
public interface IDevice : IDisposable
{
    int Id { get; }
    string Name { get; }
    Core.Models.DeviceState State { get; }

    Task ConnectAsync(CancellationToken ct = default);
    Task DisconnectAsync(CancellationToken ct = default);

    // 事件:数据到达时触发(Day 6 学过 event)
    event EventHandler<Core.Models.SensorPoint>? DataReceived;
}

// 设备基类 - "师傅":实现通用骨架,子类只填 ConnectAsync 内部细节
public abstract class DeviceBase : IDevice
{
    public int Id { get; }
    public string Name { get; }
    public Core.Models.DeviceState State { get; protected set; }

    protected DeviceBase(int id, string name)
    {
        Id = id;
        Name = name;
    }

    public abstract Task ConnectAsync(CancellationToken ct = default);
    public abstract Task DisconnectAsync(CancellationToken ct = default);

    // 子类调用:触发数据到达事件
    protected void RaiseData(Core.Models.SensorPoint point)
    {
        State = Core.Models.DeviceState.Online;
        DataReceived?.Invoke(this, point);
    }

    public event EventHandler<Core.Models.SensorPoint>? DataReceived;

    public virtual void Dispose() { }
}

⭐ 为什么抽象类 + 接口双写:IDevice 定义"能力"(可被多个不同基类实现),DeviceBase 提供"共享实现"(子类省代码)。Day 5 详细讲过。

三、报警规则(M6 用,Day 7 也提过回滞)

namespace DaqMonitor.Core.Alarming;

// 报警规则接口
public interface IAlarmRule
{
    int PointId { get; }
    Core.Models.AlarmLevel Evaluate(double value);
}

// 阈值规则 + 回滞(防止边界抖动反复触发)
public class ThresholdRule : IAlarmRule
{
    public int PointId { get; }
    public double HighThreshold { get; }     // 超过此值 → 报警
    public double LowThreshold { get; }      // 降回此值以下 → 解除
    public Core.Models.AlarmLevel TriggerLevel { get; }

    private bool _isAlarming;

    public ThresholdRule(int pointId, double high, double low,
                         Core.Models.AlarmLevel level = Core.Models.AlarmLevel.Warning)
    {
        PointId = pointId;
        HighThreshold = high;
        LowThreshold = low;
        TriggerLevel = level;
    }

    public Core.Models.AlarmLevel Evaluate(double value)
    {
        if (value > HighThreshold) _isAlarming = true;
        else if (value < LowThreshold) _isAlarming = false;
        // 中间区(回滞带):保持原状态
        return _isAlarming ? TriggerLevel : Core.Models.AlarmLevel.Normal;
    }
}

四、采集管道简化版(M0 Day 7 / M7 / M9 都引用)

真实工程里 AcquisitionPipeline 是 Channel + Timer + 批量出队的复杂版,这里给学员可编译的最小骨架:

using System.Threading.Channels;
using System.Threading.Tasks;
using DaqMonitor.Core.Models;

namespace DaqMonitor.Core.Pipeline;

// 采集管道:后台线程采集 → Channel 缓冲 → 批量事件触发
public class AcquisitionPipeline
{
    private readonly Channel<SensorPoint> _channel =
        Channel.CreateBounded<SensorPoint>(1000);

    // 外部订阅:每批数据到达时触发
    public event EventHandler<IReadOnlyList<SensorPoint>>? BatchReady;

    public void Enqueue(SensorPoint point) => _channel.Writer.TryWrite(point);

    public async Task RunAsync(System.Threading.CancellationToken ct)
    {
        await foreach (var batch in ReadBatchesAsync(ct))
            BatchReady?.Invoke(this, batch);
    }

    private async IAsyncEnumerable<IReadOnlyList<SensorPoint>> ReadBatchesAsync(
        [System.Runtime.CompilerServices.EnumeratorCancellation] System.Threading.CancellationToken ct)
    {
        var batch = new List<SensorPoint>(100);
        await foreach (var p in _channel.Reader.ReadAllAsync(ct))
        {
            batch.Add(p);
            if (batch.Count >= 100)
            {
                yield return batch;
                batch = new List<SensorPoint>(100);
            }
        }
        if (batch.Count > 0) yield return batch;
    }
}

六、ByteOrder 枚举(M2 Modbus 字节序用)

{

七、使用说明

遇到"找不到类型 X":

  1. 在本文件搜索 X,找到定义块
  2. 复制整个 namespace ... { ... } 块(或多个相关类型)
  3. 在你的项目里新建一个 .cs 文件粘贴(建议命名 _PredefinedTypes.cs)
  4. 重新编译

namespace 怎么改: - 讲义示例代码里 using DaqMonitor.Core.Models; 等 - 如果你的项目根命名空间是 MyScada,把上面所有 DaqMonitor.Core 全替换成 MyScada.Core 即可

为什么有些类型用 struct,有些用 class: - SensorPoint 用 struct(值类型,小而频繁,GC 压力小) — Day 2 学过 - SensorRecord 用 class(引用类型,EF Core 需要,有 Id 主键) - PointStat 用 record(值相等 + 不可变,M0 Day 4 学过)
