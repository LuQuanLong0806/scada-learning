一、核心数据类型(M0 Day 2 定义,几乎所有模块都用)

📂 对应真实工程 DaqMonitor.Core/Models/SensorPoint.cs

using System;

namespace DaqMonitor.Core.Models;

// 设备状态(值类型,适合状态机)
public enum DeviceState { Offline, Connecting, Online }

// 报警级别
public enum AlarmLevel { Normal, Warning, Critical }

// 采集点位 - 上位机最基础的数据载体
// struct:值类型,小而频繁传递,GC 压力小(Day 2 学过)
// 注意:这是工程真实形态 —— 公开字段 + 无构造函数,用对象初始化器赋值:
//   var p = new SensorPoint { Id = 1, Value = 20.5, Timestamp = DateTime.Now };
public struct SensorPoint
{
    public int Id;
    public double Value;
    public DeviceState State;      // 点位质量(来自设备)
    public DateTime Timestamp;     // 采样时间戳(采集源统一打,下游共用)
}

// 报警记录(Day 2 练习落地)
public struct Alarm
{
    public int PointId;
    public AlarmLevel Level;
    public double Value;
}

⚠️ new SensorPoint { ... } 不写 Timestamp 会怎样? Timestamp 落到 default(DateTime) = 0001-01-01。 所以订阅设备事件转 SensorPoint 时,一定要把 e.Timestamp 抄过来(参考工程 AcquisitionPipeline.OnPoint 就是这么干的);只有单元测试里不关心时间的场景才可以省。

二、持久化实体(M4 用,领域模型 ↔ 持久化模型分离)

📂 对应真实工程 DaqMonitor.Core/Store/SensorRecord.cs

using DaqMonitor.Core.Models;

namespace DaqMonitor.Core.Store;

// SensorPoint 是 struct,EF Core 对 struct 实体不友好 → 落库转成这个 class
// 注意两个字段名跟 SensorPoint 不一样:主键 Id(int 自增)、时间字段叫 Time(不是 Timestamp)
public class SensorRecord
{
    public int Id { get; set; }            // 自增主键(与业务 PointId 不是一回事)
    public int PointId { get; set; }       // 点位 ID(来自 SensorPoint.Id)
    public double Value { get; set; }
    public DeviceState State { get; set; }
    public DateTime Time { get; set; }     // 采样时间戳(来自 SensorPoint.Timestamp)

    public static SensorRecord FromPoint(in SensorPoint p) => new()
    { PointId = p.Id, Value = p.Value, State = p.State, Time = p.Timestamp };

    public SensorPoint ToPoint() => new()
    { Id = PointId, Value = Value, State = State, Timestamp = Time };
}

🔥 高频翻车点:SensorPoint.Timestamp vs SensorRecord.Time — 名字故意不同,提醒你"领域模型"和"落库模型"是两个东西。查历史数据用 r.Time,内存点位用 p.Timestamp。

三、设备抽象(M0 Day 5/6 定义,M1/M2/M3/M11/M13 都继承)

📂 对应真实工程 DaqMonitor.Core/Devices/IDevice.cs(同一个文件里三个类型)

using DaqMonitor.Core.Models;
using System;

namespace DaqMonitor.Core.Devices;

/// <summary>数据到达事件参数(Day 6 事件机制落地)</summary>
public class DataEventArgs : EventArgs
{
    public int PointId { get; init; }
    public double Value { get; init; }
    public DateTime Timestamp { get; init; } = DateTime.Now;   // 采集源统一打时间戳
}

/// <summary>设备统一接口:UI/采集层只认它,不关心底层是串口/网口/PLC</summary>
public interface IDevice
{
    int Id { get; }
    string Name { get; }
    DeviceState State { get; }

    void Connect();
    void Disconnect();
    double Read(int addr);
    void Write(int addr, double value);

    /// <summary>采集层拿到数据后单向通知订阅方(UI 刷新用)</summary>
    event EventHandler<DataEventArgs>? DataReceived;
}

/// <summary>设备基类:复用通用状态与事件触发逻辑</summary>
public abstract class DeviceBase : IDevice
{
    public int Id { get; }
    public string Name { get; }
    public DeviceState State { get; protected set; } = DeviceState.Offline;

    public event EventHandler<DataEventArgs>? DataReceived;

    protected DeviceBase(int id, string name) { Id = id; Name = name; }

    public abstract void Connect();
    public abstract void Disconnect();
    public abstract double Read(int addr);
    public abstract void Write(int addr, double value);

    /// <summary>子类采集到数据后调用:触发 DataReceived 事件</summary>
    protected void RaiseData(int pointId, double value)
        => DataReceived?.Invoke(this, new DataEventArgs { PointId = pointId, Value = value, Timestamp = DateTime.Now });
}

⭐ 为什么接口是同步的 Connect() 而不是 ConnectAsync()? 参考工程选了"同步门面 + 内部异步"的折衷:接口方法立即返回,真正的 IO/轮询在 Connect() 内部用 Task.Run 起(参 SimulatedDevice.Start(TimeSpan)/PlcDevice)。面试可讲:"设备接口同步语义 + 内部后台循环异步实现,上层调用简单,下层不阻塞"。C# 陷阱文档的 async 三件套(不 .Result、不 async void、token 传递)在这些后台循环里全部遵守。

四、报警引擎(M6 用)

📂 对应真实工程 DaqMonitor.Core/Alarms/AlarmRule.cs + AlarmEngine.cs + AlarmEvent.cs

using DaqMonitor.Core.Models;
using System;
using System.Collections.Generic;

namespace DaqMonitor.Core.Alarms;

/// <summary>报警规则:点位+阈值+级别+方向+回滞(生产必配回滞,防阈值抖动狂报)</summary>
public class AlarmRule
{
    public int PointId { get; set; }
    public double Threshold { get; set; }
    public AlarmLevel Level { get; set; }
    public bool IsHigh { get; set; } = true;    // true:超上限报警;false:低于下限
    public double Hysteresis { get; set; }      // 回滞带宽
}

/// <summary>报警触发/恢复事件参数</summary>
public class AlarmEvent : EventArgs
{
    public int PointId { get; init; }
    public AlarmLevel Level { get; init; }
    public double Value { get; init; }
}

/// <summary>报警引擎:规则与数据分离 + 边沿触发 + 回滞(完整实现见工程,骨架如下)</summary>
public class AlarmEngine
{
    private readonly List<AlarmRule> _rules = new();
    private readonly HashSet<int> _active = new();   // 当前已报警点位
    private readonly object _gate = new();

    public event EventHandler<AlarmEvent>? AlarmTriggered;
    public event EventHandler<AlarmEvent>? AlarmCleared;

    public void Add(AlarmRule r) { lock (_gate) _rules.Add(r); }

    public void Evaluate(SensorPoint p)
    {
        foreach (var r in _rules.ToList())
        {
            if (r.PointId != p.Id) continue;
            bool breach = r.IsHigh ? p.Value > r.Threshold : p.Value < r.Threshold;
            bool inBand = r.Hysteresis > 0 && Math.Abs(p.Value - r.Threshold) <= r.Hysteresis;
            if (breach && !inBand)
            {
                bool wasActive;
                lock (_gate) wasActive = !_active.Add(p.Id);
                if (!wasActive)   // 仅上升沿触发,不刷屏
                    AlarmTriggered?.Invoke(this, new AlarmEvent { PointId = p.Id, Level = r.Level, Value = p.Value });
            }
            else if (!breach && r.Hysteresis > 0)
            {
                bool wasActive;
                lock (_gate) wasActive = _active.Remove(p.Id);
                if (wasActive)    // 仅下降沿恢复
                    AlarmCleared?.Invoke(this, new AlarmEvent { PointId = p.Id, Level = r.Level, Value = p.Value });
            }
        }
    }
}

五、采集管道(M0 Day 7 / M5 / M6 / M7 / M9 都引用)

📂 对应真实工程 DaqMonitor.Core/Acquisition/AcquisitionPipeline.cs(简化骨架,字段名/事件签名与工程一致)

using DaqMonitor.Core.Devices;
using DaqMonitor.Core.Models;
using System.Threading.Channels;

namespace DaqMonitor.Core.Acquisition;

/// <summary>统一采集管道:所有设备事件 → Channel 缓冲 → 后台消费+定时批量出队 → BatchReady</summary>
public sealed class AcquisitionPipeline : IDisposable
{
    private readonly Channel<SensorPoint> _channel = Channel.CreateUnbounded<SensorPoint>();
    private readonly List<IDevice> _devices = new();
    private readonly CancellationTokenSource _cts = new();
    private readonly Timer _flushTimer;
    private readonly object _gate = new();
    private List<SensorPoint> _pending = new();

    /// <summary>批量就绪事件:后台线程触发,UI 订阅方需 Dispatcher 回 UI 线程</summary>
    public event EventHandler<IReadOnlyList<SensorPoint>>? BatchReady;

    public AcquisitionPipeline(TimeSpan flushInterval, int maxBatch = 500)
        => _flushTimer = new Timer(_ => Flush(), null, flushInterval, flushInterval);

    /// <summary>注册设备:自动订阅它的 DataReceived,把点塞进缓冲</summary>
    public void Register(IDevice device)
    {
        device.DataReceived += OnPoint;
        _devices.Add(device);
    }

    // ⚠️ 关键三行:DataEventArgs → SensorPoint 的转换(必须抄 e.Timestamp!)
    private void OnPoint(object? sender, DataEventArgs e)
        => _channel.Writer.TryWrite(new SensorPoint { Id = e.PointId, Value = e.Value, Timestamp = e.Timestamp });

    private void Flush()
    {
        List<SensorPoint>? batch = null;
        lock (_gate) { if (_pending.Count > 0) { batch = _pending; _pending = new(); } }
        if (batch is not null) BatchReady?.Invoke(this, batch);
    }

    public void Dispose()
    {
        _cts.Cancel(); _flushTimer.Dispose(); _channel.Writer.TryComplete();
        foreach (var d in _devices) d.DataReceived -= OnPoint;
    }
}

💡 管道构造即启动(定时器/消费循环自己跑),没有 Start()/StopAsync() 方法——控制采集的开关在设备侧(device.Connect()/Disconnect() 或 SimulatedDevice.Start/Stop)。

八、使用说明

遇到"找不到类型 X":

  1. 在本文件搜索 X,找到定义块
  2. 复制整个 namespace ... { ... } 块(或多个相关类型)
  3. 在你的项目里新建一个 .cs 文件粘贴(建议命名 _PredefinedTypes.cs)
  4. 重新编译

namespace 怎么改: - 讲义示例代码里 using DaqMonitor.Core.Models; 等 - 如果你的项目根命名空间是 MyScada,把上面所有 DaqMonitor.Core 全替换成 MyScada.Core 即可

为什么有些类型用 struct,有些用 class: - SensorPoint/Alarm 用 struct(值类型,小而频繁,GC 压力小) — Day 2 学过 - SensorRecord 用 class(引用类型,EF Core 需要,有 Id 主键) - PointStat 用 record(值相等 + 不可变,M0 Day 4 学过)
